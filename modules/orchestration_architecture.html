from mcp.server.fastmcp import FastMCP
from core.orchestration.ingestion import fetch_news
from core.orchestration.processor import analyze_data

# 1. Initialize the AI-Ready Orchestrator
mcp = FastMCP("NewsIntelligence")

@mcp.tool()
async def run_news_pipeline(topic: str):
    """
    Full Modular Pipeline: Ingestion -> Analysis -> Result.
    """
    # Fetch data (Async)
    raw_data = await fetch_news(topic)
    
    # Clean & Process (Decoupled Logic)
    final_results = analyze_data(raw_data)
    
    return final_results

if __name__ == "__main__":
    mcp.run()
