Code Example: Analysis & Viz

This script demonstrates a statistical correlation test combined with a visual regression analysis.

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from scipy.stats import pearsonr

# 1. Loading & Exploration
df = sns.load_dataset('tips')

# 2. Statistical Correlation Test
corr, p_value = pearsonr(df['total_bill'], df['tip'])
print(f"Correlation: {corr:.2f}")

# 3. Visualization with Seaborn
plt.figure(figsize=(10, 6))
sns.regplot(x='total_bill', y='tip', data=df, color='white')
plt.title("Relationship: Bill vs Tip (Regression Line)")
plt.show()
